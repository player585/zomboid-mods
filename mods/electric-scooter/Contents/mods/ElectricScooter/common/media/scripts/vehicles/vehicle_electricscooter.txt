/** ============================================================
    Electric Scooter - Vehicle Script (Build 42)
    Mod: ElectricScooter
    Author: zmodder

    A near-silent, battery-powered single-seater. Treats the
    car battery as its only fuel source (no gas tank). Spawns
    rare in suburb / parking pools.

    Uses B42 vehicle template inheritance (template! = SmallCar)
    instead of declaring every part inline. Overrides only the
    pieces that differ from a small vanilla car.
    ============================================================ **/

module Base
{

    vehicle ElectricScooter
    {
        /** Pull in the small-car chassis: areas, wheels, seats,
            engine, battery, tires, brakes, doors, windows, sounds,
            collision. We then override what matters. **/
        template! = SmallCar,

        /** ---- Identity / handling ---- **/
        mechanicType            = 1,
        engineRepairLevel       = 2,
        playerDamageProtection  = 0.25,

        /** ---- Performance: light, slow, near-silent ---- **/
        mass                    = 90,
        engineForce             = 2800,
        engineQuality           = 60,
        engineLoudness          = 4,
        maxSpeed                = 50,
        brakingForce            = 40,
        steeringIncrement       = 0.04,
        steeringClamp           = 0.3,
        stoppingMovementForce   = 2.0,
        rollInfluence           = 0.15,
        wheelFriction           = 1.2,
        offRoadEfficiency       = 0.65,

        /** ---- Custom skin / damage textures ---- **/
        textureMask             = vehicles/ElectricScooter_mask,
        textureLights           = vehicles/ElectricScooter_lights,
        textureRust             = vehicles/ElectricScooter_rust,
        textureDamage1Overlay   = vehicles/ElectricScooter_damage1,
        textureDamage2Overlay   = vehicles/ElectricScooter_damage2,

        /** ---- 3D model (placeholder cube for now) ---- **/
        model
        {
            file    = vehicles/ElectricScooter,
            scale   = 0.012,
            offset  = 0 0 0,
        }

        /** ---- Custom near-silent sound set ---- **/
        sound
        {
            engine          = ElectricScooterHum,
            engineStart     = ElectricScooterStart,
            engineTurnOff   = ElectricScooterOff,
            horn            = VehicleHornStandard,
            ignitionFail    = VehicleIgnitionFailSmallCar,
        }

        /** ---- Part overrides ---- **/

        /** Battery: vanilla CarBattery, but tagged via Lua hook
            to drain even when stopped (handled by our update fn). **/
        part Battery
        {
            itemType = Base.CarBattery,
            lua
            {
                create = Vehicles.Create.Battery,
                update = ElectricScooter.Update.Battery,
            }
        }

        /** Engine: keep vanilla install/uninstall flow, but our
            update function locks fuel at max so the vanilla "no gas"
            check never fires (scooter has no gas tank). **/
        part Engine
        {
            lua
            {
                create      = Vehicles.Create.Engine,
                update      = ElectricScooter.Update.Engine,
                checkEngine = Vehicles.CheckEngine.Engine,
            }
        }

        /** Hide the (inherited) gas tank visually — we keep the part
            so vanilla code that probes for it doesn't crash, but lock
            it full and uninteractable. **/
        part GasTank
        {
            category = nodisplay,
        }

    } /** end vehicle ElectricScooter **/


    /** ====================================================
        Custom sound declarations (referenced above)
        ==================================================== **/

    sound ElectricScooterHum
    {
        category    = Vehicle,
        clip
        {
            file            = media/sound/ElectricScooterHum.ogg,
            distanceMax     = 15,
            distanceMin     = 1,
            volume          = 0.35,
            pitch           = 1.0,
        }
    }

    sound ElectricScooterStart
    {
        category    = Vehicle,
        clip
        {
            file            = media/sound/ElectricScooterStart.ogg,
            distanceMax     = 10,
            volume          = 0.4,
        }
    }

    sound ElectricScooterOff
    {
        category    = Vehicle,
        clip
        {
            file            = media/sound/ElectricScooterOff.ogg,
            distanceMax     = 8,
            volume          = 0.3,
        }
    }

} /** end module Base **/
